<script>
  let totalBaseline = 0, totalPruned = 0, totalQuestions = 0;
  let history = [];

  function askSuggestion(q) {
    document.getElementById('query-input').value = q;
    sendMessage();
  }

  function sendMessage() {
    const input = document.getElementById('query-input');
    const query = input.value.trim();
    if (!query) return;
    input.value = '';

    document.getElementById('empty-state').style.display = 'none';
    appendUserMsg(query);
    const typingEl = appendTyping();

    fetch('/ask', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ query })
    })
    .then(r => r.json())
    .then(data => {
      typingEl.remove();
      if (data.error) {
        appendBotMsg('Error: ' + data.error, [], 0, 0, 0);
        return;
      }
      appendBotMsg(data.answer, data.sources, data.saving_pct, data.latency_ms, data.baseline_tok);
      updateStats(data.baseline_tok, data.pruned_tok, data.saving_pct, query, data.sources);
    })
    .catch(e => {
      typingEl.remove();
      appendBotMsg('Something went wrong. Please try again.', [], 0, 0, 0);
    });
  }

  function appendUserMsg(text) {
    const chat = document.getElementById('chat-area');
    const div = document.createElement('div');
    div.className = 'msg-user';
    div.innerHTML = `<div class="bubble">${escHtml(text)}</div>`;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
  }

  function appendTyping() {
    const chat = document.getElementById('chat-area');
    const div = document.createElement('div');
    div.className = 'typing';
    div.innerHTML = `
      <div class="bot-avatar">ET</div>
      <div class="typing-dots"><span class="dot"></span><span class="dot"></span><span class="dot"></span></div>`;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
    return div;
  }

  function appendBotMsg(text, sources, savingPct, latencyMs, baselineTok) {
    const chat = document.getElementById('chat-area');
    const div = document.createElement('div');
    div.className = 'msg-bot';

    let sourceBadges = sources.map(s =>
      `<span class="badge badge-blue">📖 ${escHtml(s)}</span>`
    ).join('');
    let metaHtml = savingPct > 0 ? `
      ${sourceBadges}
      <span class="badge badge-green">💰 ${savingPct}% saved</span>
      <span class="badge badge-orange">⚡ ${latencyMs} ms</span>
    ` : '';

    div.innerHTML = `
      <div class="bot-avatar">ET</div>
      <div class="bot-bubble">
        <div class="bot-text">${escHtml(text)}</div>
        ${metaHtml ? `<div class="bot-meta">${metaHtml}</div>` : ''}
      </div>`;
    chat.appendChild(div);
    chat.scrollTop = chat.scrollHeight;
  }

  function updateStats(baseline, pruned, savingPct, query, sources) {
    totalBaseline  += baseline;
    totalPruned    += pruned;
    totalQuestions += 1;

    const overallSaving = totalBaseline > 0
      ? Math.round((1 - totalPruned / totalBaseline) * 100)
      : 0;

    document.getElementById('stat-questions').textContent = totalQuestions;
    document.getElementById('stat-baseline').textContent  = totalBaseline.toLocaleString();
    document.getElementById('stat-pruned').textContent    = totalPruned.toLocaleString();
    document.getElementById('stat-saving').textContent    = overallSaving + '%';
    document.getElementById('saving-bar').style.width     = overallSaving + '%';
    document.getElementById('bar-label').textContent      = overallSaving + '%';

    history.unshift({ query, savingPct, source: sources[0] || '—' });
    renderHistory();
  }

  function renderHistory() {
    const list = document.getElementById('history-list');
    list.innerHTML = history.slice(0, 10).map(h => `
      <div class="history-item" onclick="askSuggestion('${escHtml(h.query)}')">
        <div class="history-q">${escHtml(h.query)}</div>
        <div class="history-meta">${escHtml(h.source)} · ${h.savingPct}% saved</div>
      </div>
    `).join('');
  }

  function clearChat() {
    const chat = document.getElementById('chat-area');
    chat.innerHTML = '';
    const empty = document.createElement('div');
    empty.className = 'empty-state';
    empty.id = 'empty-state';
    empty.innerHTML = `
      <div class="empty-icon">📚</div>
      <div class="empty-title">Ask anything from your textbooks</div>
      <div class="empty-sub">I'll find the most relevant content and explain it clearly.<br>Curriculum-aligned answers only — no hallucinations.</div>
      <div class="suggestions">
        <div class="suggestion-chip" onclick="askSuggestion('What is inertia?')">What is inertia?</div>
        <div class="suggestion-chip" onclick="askSuggestion('Explain Newton\\'s laws')">Newton's laws</div>
        <div class="suggestion-chip" onclick="askSuggestion('What is photosynthesis?')">Photosynthesis</div>
        <div class="suggestion-chip" onclick="askSuggestion('Quadratic formula')">Quadratic formula</div>
        <div class="suggestion-chip" onclick="askSuggestion('How does phloem work?')">How does phloem work?</div>
        <div class="suggestion-chip" onclick="askSuggestion('What is Ohm\\'s law?')">Ohm's law</div>
      </div>`;
    chat.appendChild(empty);
    totalBaseline = totalPruned = totalQuestions = 0;
    history = [];
    renderHistory();
    document.getElementById('stat-questions').textContent = '0';
    document.getElementById('stat-baseline').textContent  = '0';
    document.getElementById('stat-pruned').textContent    = '0';
    document.getElementById('stat-saving').textContent    = '0%';
    document.getElementById('saving-bar').style.width     = '0%';
    document.getElementById('bar-label').textContent      = '0%';
  }

  function escHtml(str) {
    return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }
</script>
